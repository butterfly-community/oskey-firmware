cmake_minimum_required(VERSION 3.20)
project(oskey_nxp_tests C)
enable_testing()
option(NXP_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(NXP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(PSA REQUIRED IMPORTED_TARGET mbedcrypto)
set(root ${CMAKE_CURRENT_LIST_DIR}/../../..)
set(nxp ${root}/src/security/nxp)
set(nano ${root}/lib/nano-package/lib)
add_library(nxp_crypto STATIC ${nxp}/host_crypto_psa.c)
target_include_directories(nxp_crypto PUBLIC ${nxp} ${nxp}/port ${nano}/apdu ${nano}/apdu/scp03 ${nano}/t1oi2c)
target_compile_definitions(nxp_crypto PUBLIC OSKEY_NXP_HOST_TEST CONFIG_PLUGANDTRUST_APDU_BUFFER_SIZE=255 WITH_PLATFORM_SCP03)
target_compile_options(nxp_crypto PRIVATE -Wall -Wextra -Werror)
target_link_libraries(nxp_crypto PUBLIC PkgConfig::PSA pthread)
foreach(suite crypto store device faults scp03)
  add_executable(test_${suite} test_${suite}.c)
  target_compile_options(test_${suite} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(test_${suite} PRIVATE nxp_crypto)
  add_test(NAME nxp_${suite} COMMAND test_${suite})
endforeach()
target_sources(test_scp03 PRIVATE ${nano}/apdu/scp03/se05x_scp03.c
  ${nano}/apdu/scp03/se05x_auth_utils.c)
target_compile_options(test_scp03 PRIVATE -ffunction-sections -fdata-sections)
target_link_options(test_scp03 PRIVATE -Wl,--gc-sections)
target_sources(test_store PRIVATE ${nxp}/store.c)
target_sources(test_device PRIVATE ${nxp}/device.c)
foreach(function psa_import_key psa_destroy_key psa_generate_random psa_mac_compute
    psa_mac_sign_setup psa_mac_update psa_mac_sign_finish psa_cipher_encrypt_setup
    psa_cipher_decrypt_setup psa_cipher_set_iv psa_cipher_update psa_cipher_finish)
  target_link_options(test_faults PRIVATE -Wl,--wrap=${function})
endforeach()
